let dragState=null;
function startDrag(ev){
  const id=ev.currentTarget.dataset.id;
  if(!id)return;
  const e=cur().elements.find(x=>x.id===id);
  if(!e)return;
  state.selected=id;
  checkpoint();
  dragState={id,startX:ev.clientX,startY:ev.clientY,origX:e.x,origY:e.y};
  ev.currentTarget.setPointerCapture?.(ev.pointerId);
  ev.preventDefault();
}
function moveDrag(ev){
  if(!dragState)return;
  const e=cur().elements.find(x=>x.id===dragState.id);
  if(!e)return;
  e.x=Math.max(15,Math.min(1135-e.w,dragState.origX+(ev.clientX-dragState.startX)));
  e.y=Math.max(10,Math.min(745-e.h,dragState.origY+(ev.clientY-dragState.startY)));
  const el=ev.currentTarget;
  el.style.left=e.x+'px';el.style.top=e.y+'px';
  const ins=$('inspector'); if(ins&&sel()?.id===e.id){const x=$('x');const y=$('y');if(x)x.value=e.x;if(y)y.value=e.y}
}
function endDrag(){if(dragState){log('Moved element');render();}dragState=null}
function bindCanvasEditing(){
  document.querySelectorAll('#canvas .el').forEach(el=>{
    const nodeId=el.getAttribute('onclick')?.match(/pick\(event,'([^']+)'\)/)?.[1];
    if(nodeId)el.dataset.id=nodeId;
    el.addEventListener('pointerdown',startDrag);
    el.addEventListener('pointermove',moveDrag);
    el.addEventListener('pointerup',endDrag);
    el.addEventListener('pointercancel',endDrag);
  });
}
